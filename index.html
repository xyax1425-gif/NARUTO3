const button = document.getElementById("draw"); const result = document.getElementById("result"); const title = document.getElementById("resultTitle"); const text = document.getElementById("resultText"); const ticketNumber = document.getElementById("ticketNumber"); const burst = document.getElementById("burst"); button.addEventListener("click", async () => { const number = ticketNumber.value.trim(); if (!number) { alert("抽選番号を入力してください！"); ticketNumber.focus(); return; } if (!/^\d{1,20}$/.test(number)) { alert("番号は1〜20桁の数字で入力してください！"); return; } button.disabled = true; ticketNumber.disabled = true; button.textContent = "抽選中…"; burst.classList.add("on"); try { const response = await fetch("/.netlify/functions/draw", { method: "POST", headers: { "Content-Type": "application/json" }, body: JSON.stringify({ number }) }); const data = await response.json(); if (!response.ok) { throw new Error(data.error || "抽選できませんでした"); } title.textContent = data.win ? "大当たり！景品GET！" : "今回ははずれ…"; text.textContent = data.win ? "おめでとうございます！景品をお受け取りください！" : "参加ありがとう！またの挑戦をお待ちしています。"; result.classList.add("show"); button.textContent = "抽選終了"; } catch (error) { alert(error.message); button.disabled = false; ticketNumber.disabled = false; button.textContent = "運命の抽選をする！"; } finally { burst.classList.remove("on"); } });
